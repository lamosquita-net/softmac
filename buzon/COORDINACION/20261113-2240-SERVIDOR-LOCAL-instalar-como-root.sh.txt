#!/bin/bash
# Sync 3.2 H2, pasos que exigen root en ns2. Preparado por SERVIDOR-LOCAL-3 (13-11-2026) por orden de COORDINACIÓN-41.
# Quién lo ejecuta: Titín, en ns2, con su usuario con sudo. NO muestra ningún valor secreto.
#
#   bash instalar-como-root.sh binario        # paso 1: instala el binario nuevo (el servicio sigue como hasta ahora)
#   bash instalar-como-root.sh credenciales   # paso 2: crea cuenta-hmac y cuenta-aes (solo cuando SERVIDOR-LOCAL lo pida)
#
# Nada se borra: el binario anterior queda como flyweb-sync.antes-h2 en la misma carpeta.
set -euo pipefail

SHA=1a8362172b34d1434565e553e85f132fd5b2158f97dd8149096ab54b3e21cc2a
URL=https://github.com/lamosquita-net/softmac/releases/download/flyweb-sync/flyweb-sync
DEST=/opt/flyweb-sync

case "${1:-}" in
binario)
  T=$(mktemp -d)
  cd "$T"
  curl -fsSLo flyweb-sync "$URL"
  echo "$SHA  flyweb-sync" | sha256sum -c -        # si no coincide, el script se para aquí
  sudo cp -p "$DEST/flyweb-sync" "$DEST/flyweb-sync.antes-h2"
  sudo install -m 0755 -o root -g root flyweb-sync "$DEST/flyweb-sync"
  sudo sha256sum "$DEST/flyweb-sync"                # tiene que ser $SHA
  sudo ls -l "$DEST"
  echo "Binario instalado. El servicio no se ha reiniciado: sigue con el anterior hasta que SERVIDOR-LOCAL lo reinicie."
  ;;
credenciales)
  for n in cuenta-hmac cuenta-aes; do
    if sudo test -e "/etc/credstore/$n"; then echo "ya existe /etc/credstore/$n: no se toca; parar y avisar"; exit 1; fi
  done
  # Formato del README: una línea «versión clave_hex» (32 bytes). Los valores se generan y se escriben dentro de sudo;
  # no se imprimen ni quedan en variables de este shell.
  sudo sh -c 'umask 077; for n in cuenta-hmac cuenta-aes; do printf "1 %s\n" "$(openssl rand -hex 32)" > "/etc/credstore/$n"; done'
  sudo ls -l /etc/credstore/cuenta-hmac /etc/credstore/cuenta-aes   # root 0600, 69 bytes cada una (sin mostrar el contenido)
  echo "Credenciales creadas (root 0600). Avisa a SERVIDOR-LOCAL."
  ;;
*)
  echo "Uso: bash $0 binario | credenciales"; exit 2 ;;
esac
